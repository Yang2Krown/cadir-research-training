info <- data.frame(
  Biclusters = as.factor(data$CAdir),
  Truth = as.character(data$Annotation)
) %>%
  as_tibble() %>%
  group_by(Biclusters, Truth) %>%
  summarise(n = n()) %>%
  group_by(Truth) %>%
  mutate(Partition = n / sum(n)) %>%
  ungroup() %>%
  as.data.frame() %>%
  group_by(Truth) %>%
  filter(Partition == max(Partition)) %>%
